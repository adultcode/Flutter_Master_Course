void main() {
  runApp( MyApp());
}

class MyApp extends StatefulWidget {


  @override
  State<MyApp> createState() => _MyAppState();
}

class _MyAppState extends State<MyApp> {

  var _value = 0.0;


  @override
  Widget build(BuildContext context) {
    return MaterialApp(
        debugShowCheckedModeBanner: false,
        home: Scaffold(
          appBar: AppBar(
            backgroundColor: Color(0xfff7c307),
            title: Text("Hello"),
            centerTitle: true,

          ),


          body: Center(
            child: Column(
              children: [
                Slider(
                  min: 0.0,
                  max: 100.0,
                  value: _value,
                  onChanged: (value) {
                    setState(() {
                      _value = value;
                    });
                  },
                ),
                SizedBox(height: 30.0,),
                Text(_value.toString())
              ],
            ),
          ),


        )
    );
  }
}
